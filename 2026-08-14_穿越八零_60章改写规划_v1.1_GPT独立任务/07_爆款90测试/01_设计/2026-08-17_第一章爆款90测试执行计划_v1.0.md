### Task 2: 建立正文机械验收器

- [ ] **Step 1: 新建验证脚本**

脚本完整内容：

```powershell
param([Parameter(Mandatory=$true)][string]$BodyPath)
$ErrorActionPreference='Stop'
[Console]::OutputEncoding=New-Object System.Text.UTF8Encoding($false)
if(-not (Test-Path -LiteralPath $BodyPath -PathType Leaf)){throw "File not found: $BodyPath"}
$lines=@(Get-Content -LiteralPath $BodyPath -Encoding UTF8)
$raw=[string]::Join("`n",$lines)
$errors=New-Object System.Collections.Generic.List[string]
$heading=-1
for($i=0;$i -lt $lines.Count;$i++){if($lines[$i] -match '^##\s+第一章\s+'){ $heading=$i; break }}
if($heading -lt 0){[void]$errors.Add('Chapter 1 heading missing');$body=$raw}else{
  $tail=@($lines[($heading+1)..($lines.Count-1)])
  $stop=$tail.Count
  for($i=0;$i -lt $tail.Count;$i++){if($tail[$i] -eq '---' -or $tail[$i] -match '^##\s+第一章状态增量'){ $stop=$i; break }}
  if($stop -le 0){$body=''}else{$body=[string]::Join("`n",$tail[0..($stop-1)])}
}
$han=[regex]::Matches($body,'[\u4e00-\u9fff]')
$hanCount=$han.Count
if($hanCount -lt 2100 -or $hanCount -gt 2300){[void]$errors.Add("Han count outside 2100-2300: $hanCount")}
$firstCount=[Math]::Min(300,$han.Count)
$firstHan=if($firstCount){[string]::Join('',@($han|Select-Object -First $firstCount|ForEach-Object{$_.Value}))}else{''}
foreach($anchor in @('存折','自行车','姜远')){if(-not $firstHan.Contains($anchor)){[void]$errors.Add("First-300 anchor missing: $anchor")}}
foreach($name in @('姜瑶','沈桂兰','沈建勋','陈丽')){if(-not $body.Contains($name)){[void]$errors.Add("Character missing: $name")}}
if($raw -match '(?m)^##\s+第二章'){[void]$errors.Add('Unexpected Chapter 2 content')}
if($body.Contains('沈笑笑必须跟') -or $body.Contains('笑笑必须跟')){[void]$errors.Add('Old escort hook remains')}
$paras=@($body -split "(?:`r?`n){2,}"|Where-Object{$_.Trim()})
$last=if($paras.Count){$paras[-1]}else{''}
foreach($anchor in @('沈建勋','自行车','柳河')){if(-not $last.Contains($anchor)){[void]$errors.Add("Final-risk anchor missing: $anchor")}}
$hash=(Get-FileHash -LiteralPath $BodyPath -Algorithm SHA256).Hash
[pscustomobject]@{Status=if($errors.Count){'FAIL'}else{'PASS'};Path=$BodyPath;HanCount=$hanCount;Paragraphs=$paras.Count;First300Han=$firstHan;LastParagraph=$last;ErrorCount=$errors.Count;Errors=@($errors);SHA256=$hash}|ConvertTo-Json -Depth 4
if($errors.Count){exit 1}
```
